# Recaply iOS — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Ship Recaply, a native iOS meeting/lecture summarizer with a 4-stage on-device+cloud AI pipeline and a cinematic UI.

**Architecture:** SwiftUI MVVM (`ObservableObject` + `@Published`, no `@Observable`). Services with one job each, constructor-injected, communicating through value-type domain models. Pipeline per recording: Capture → Transcribe (Speech, on-device) → Classify (CoreML, on-device) → Enhance (LLM) → Summarize (LLM), persisted in Core Data.

**Tech Stack:** Swift 5.0, iOS 16.4, Xcode 14.3.1 (objectVersion 56). System frameworks only: AVFoundation, Speech, CoreML, Natural Language (optional), Core Data, SwiftUI (`NavigationStack`), Charts (stretch). Cloud LLM via `URLSession` (GLM primary, Deepseek fallback, OpenAI-compatible).

This project is **authored on Linux/WSL without Xcode**. Swift that touches iOS frameworks (AVFoundation, Speech, CoreML, CoreData, SwiftUI) **cannot compile or run here**. Therefore:

- **Compile and run XCTest** at the iOS lab (Mac). The "Run test" steps below are written as TDD discipline (write test → write impl) but their *execution* is a lab activity; expected results are documented so the lab run is unambiguous.

Never use an iOS 17+ API. In particular: no `@Observable` macro (use `ObservableObject` + `@Published`), no Apple Translation framework.

- [ ] **Step 1: Write `RecordingService`**

```swift
import Foundation
import AVFoundation

protocol RecordingProviding {
    func start() throws -> URL
    func stop() throws -> (url: URL, duration: TimeInterval)
    var currentLevel: Float { get }  // 0...1 for the waveform
}

final class RecordingService: RecordingProviding {
    static let shared = RecordingService()

    private let engine = AVAudioEngine()
    private var file: AVAudioFile?
    private var fileURL: URL?
    private var startTime: Date?
    private var levelTimer: Timer?
    private(set) var currentLevel: Float = 0

    func start() throws -> URL {
        let session = AVAudioSession.sharedInstance()
        try session.setCategory(.playAndRecord, mode: .default, options: [.defaultToSpeaker])
        try session.setActive(true)

        let url = FileManager.default
            .urls(for: .documentDirectory, in: .userDomainMask)[0]
            .appendingPathComponent("rec-\(UUID().uuidString).m4a")
        let settings: [String: Any] = [
            AVFormatIDKey: kAudioFormatMPEG4AAC,
            AVSampleRateKey: 44100, AVNumberOfChannelsKey: 1
        ]
        engine.inputNode.removeTap(onBus: 0)
        engine.inputNode.installTap(onBus: 0, bufferSize: 1024) { [weak self] buffer, _ in
            guard let self else { return }
            self.currentLevel = self.rms(buffer)
            if self.file == nil {
                self.file = try? AVAudioFile(forWriting: url, settings: settings)
            }
            try? self.file?.write(from: buffer)
        }
        try engine.start()
        startTime = Date()
        fileURL = url
        return url
    }

    func stop() throws -> (url: URL, duration: TimeInterval) {
        engine.inputNode.removeTap(onBus: 0)
        engine.stop()
        try AVAudioSession.sharedInstance().setActive(false, options: .notifyOthersOnDeactivation)
        let duration = Date().timeIntervalSince(startTime ?? Date())
        let url = fileURL ?? URL(fileURLWithPath: "/dev/null")
        file = nil; fileURL = nil
        return (url, duration)
    }

    private func rms(_ buffer: AVAudioPCMBuffer) -> Float {
        guard let data = buffer.floatChannelData?[0] else { return 0 }
        var sum: Float = 0
        for i in 0..<Int(buffer.frameLength) { sum += data[i] * data[i] }
        return sqrt(sum / Float(buffer.frameLength))
    }
}
```

```swift
import Foundation
import Combine
import AVFoundation

@MainActor
final class RecordViewModel: ObservableObject {
    enum Phase: Equatable { case idle, recording, stopping }

    @Published var phase: Phase = .idle
    @Published var level: Float = 0
    @Published var tag: SessionTag = .meeting
    @Published var title: String = ""
    @Published var cameraOn: Bool = false
    @Published var lastRecording: RecordingInfo?

    private let recorder: RecordingProviding
    private let camera: CameraService
    private var levelTimer: Timer?

    init(recorder: RecordingProviding = RecordingService.shared,
         camera: CameraService = .shared) {
        self.recorder = recorder
        self.camera = camera
    }

    func start(tag: SessionTag, title: String?) throws {
        _ = try recorder.start()
        if cameraOn { try? camera.start() }
        self.tag = tag
        self.title = title ?? ""
        phase = .recording
        levelTimer = Timer.scheduledTimer(withTimeInterval: 0.05, repeats: true) { [weak self] _ in
            Task { @MainActor in self?.level = self?.recorder.currentLevel ?? 0 }
        }
    }

    func stop() {
        levelTimer?.invalidate(); levelTimer = nil
        phase = .stopping
```

```swift
import Foundation

enum TranscriptSegmenter {
    static func segment(_ pieces: [(text: String, time: TimeInterval)]) -> [TranscriptSegmentModel] {
        var out: [TranscriptSegmentModel] = []
        var buffer = ""
        var bufferTime: TimeInterval = 0
        var index = 0
        let terminals: Set<Character> = [".", "!", "?", "。", "！", "？"]

        for piece in pieces {
            if buffer.isEmpty { bufferTime = piece.time }
            buffer += buffer.isEmpty ? piece.text : " " + piece.text
            if piece.text.last.map({ terminals.contains($0) }) == true {
                let text = buffer.trimmingCharacters(in: .whitespacesAndNewlines)
                if !text.isEmpty {
                    out.append(TranscriptSegmentModel(index: index, text: text, timestamp: bufferTime))
                    index += 1
                }
                buffer = ""
            }
        }
        let tail = buffer.trimmingCharacters(in: .whitespacesAndNewlines)
        if !tail.isEmpty {
            out.append(TranscriptSegmentModel(index: index, text: tail, timestamp: bufferTime))
        }
        return out
    }
}
```

```swift
final class TranscriptEnhancementService {
    static let shared = TranscriptEnhancementService()

            let raw = try await DeepseekClient.shared.complete(messages: msgs, jsonMode: true)

        }
    }
```

```swift
import Foundation
import Combine

@MainActor
final class PipelineCoordinator: ObservableObject {
    enum Stage: Int, CaseIterable { case transcribing, classifying, enhancing, summarizing
        var title: String {
            switch self { case .transcribing: "Transcribing"; case .classifying: "Classifying";
            case .enhancing: "Enhancing"; case .summarizing: "Summarizing" } } }

    @Published var activeStage: Stage?
    @Published var completed: Set<Stage> = []
    @Published var failedStage: Stage?
    @Published var resultRecordingID: UUID?

    private let transcriber: TranscriptionProviding
    private let classifier: ClassificationService
    private let enhancer: TranscriptEnhancementService
    private let summarizer: SummarizationService
    private let repo: RecordingRepository

    init(transcriber: TranscriptionProviding = TranscriptionService.shared,
         classifier: ClassificationService = .shared,
         enhancer: TranscriptEnhancementService = .shared,
         summarizer: SummarizationService = .shared,
         repo: RecordingRepository = .shared) { /* assign all */ }

    func run(_ recording: RecordingInfo) async {
        var info = recording
        do {
            activeStage = .transcribing
            let segments = try await transcriber.transcribe(audioAt: recording.audioURL!)
            info.status = .transcribing; repo.update(info)

            activeStage = .classifying; let labeled = classifier.classify(segments)
            repo.appendSegments(labeled, to: info); info.status = .classifying
            completed.insert(.transcribing); completed.insert(.classifying)

            activeStage = .enhancing
            let enhanced = (try? await enhancer.enhance(segments: labeled, tag: recording.tag))
                ?? EnhancedTranscriptPayload(clean: labeled.map { .init(index: $0.index, text: $0.text) },
                                             polished: [.init(heading: "Transcript",
                                                              body: labeled.map(\.text).joined(separator: " "))])
            repo.saveEnhancement(enhanced, to: info); info.status = .enhancing
            completed.insert(.enhancing)

            activeStage = .summarizing
            if let summary = try? await summarizer.summarize(segments: labeled, tag: recording.tag) {
                repo.saveSummary(summary, to: info)
            }
            info.status = .ready; repo.update(info)
            completed.insert(.summarizing); activeStage = nil
            resultRecordingID = info.id
        } catch {
            failedStage = activeStage; info.status = .failed; repo.update(info)
        }
    }
}
```
